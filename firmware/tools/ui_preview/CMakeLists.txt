cmake_minimum_required(VERSION 3.20)
project(tab5_preview C CXX)
get_filename_component(APP "${CMAKE_CURRENT_SOURCE_DIR}/../../main" ABSOLUTE)
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${APP}/../managed_components/lvgl__lvgl lvgl)
file(GLOB FONTS ${APP}/fonts/*.c)
add_executable(preview preview.c ${FONTS} ${APP}/term/vt.c ${APP}/term/term_view.c ${APP}/ui/ui_screensaver.c ${APP}/ui/ui_boot.c ${APP}/ui/ui_core.c ${APP}/ui/ui_motion.c ${APP}/ui/ui_theme.c ${APP}/ui/ui_home.c ${APP}/ui/ui_camera.c ${APP}/ui/ui_settings.c ${APP}/ui/ui_appearance.c ${APP}/i18n/i18n.c)
target_include_directories(preview PRIVATE stubs ${APP})
target_compile_options(preview PRIVATE -Wno-deprecated-declarations)
target_link_libraries(preview lvgl)
